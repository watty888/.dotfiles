# Essential packages for a fresh Arch install on the HP Pavilion Gaming 17-cd0xxx.
# Consumed by restore.sh. Comments and blank lines are ignored.
#
# NOT LISTED ON PURPOSE:
#   nvidia / nvidia-lts / nvidia-utils
#     The GTX 1050 Max-Q is Pascal. NVIDIA's 590 branch dropped Maxwell/Pascal/
#     Volta and Arch's `nvidia` tracks the current branch, so those packages do
#     not drive this GPU. The working option is nvidia-580xx-dkms from the AUR:
#         ./restore.sh --with-nvidia
#     Only needed if the external HDMI port is wired to the dGPU. Test by
#     plugging a monitor in and running:
#         for c in /sys/class/drm/card[0-9]; do
#           echo "$(basename $c) -> $(basename $(readlink -f $c/device/driver))"
#           for k in $c-*/; do echo "   $(basename $k) $(cat $k/status)"; done
#         done
#     If HDMI-A shows "connected" under the i915 card, skip NVIDIA entirely.
